⛔ النشر العام معطّل — يجب إعداد Supabase
قاعدة البيانات ترفض الكتابة. نفّذ هذا الـ SQL في لوحة تحكم Supabase → SQL Editor:
-- الصق هذا في Supabase > SQL Editor > Run
ALTER TABLE characters ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "amazad_allow_all" ON characters;
CREATE POLICY "amazad_allow_all" ON characters FOR ALL USING (true) WITH CHECK (true);

ALTER TABLE character_groups ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "amazad_allow_all" ON character_groups;
CREATE POLICY "amazad_allow_all" ON character_groups FOR ALL USING (true) WITH CHECK (true);
